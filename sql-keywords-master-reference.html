1. Core Query Keywords

KeywordExampleKya karta hai
SELECTSELECT name FROM users;Data retrieve/read karne ke liye — kaunse columns chahiye batata hai
*SELECT * FROM users;"Sabhi columns" ka symbol — pura row le aata hai
FROMSELECT * FROM users;Kis table se data lena hai, wo specify karta hai
ASSELECT name AS username FROM users;Column/table ko temporary alias (naya naam) deta hai
DISTINCTSELECT DISTINCT city FROM users;Duplicate values hata ke sirf unique results deta hai
ORDER BYSELECT * FROM users ORDER BY name;Result ko sort karta hai (default ascending)
ASC / DESCORDER BY age DESC;Sorting direction — ascending (chhota→bada) ya descending (bada→chhota)
LIMITSELECT * FROM users LIMIT 5;Sirf itne rows return karo (MySQL/PostgreSQL)
OFFSETSELECT * FROM users LIMIT 5 OFFSET 10;Pehle N rows skip karo, phir LIMIT apply karo
TOPSELECT TOP 5 * FROM users;MSSQL ka LIMIT jaisa — sirf itne rows do
FETCHSELECT * FROM users FETCH FIRST 5 ROWS ONLY;Oracle/PostgreSQL — rows limit karne ka standard tarika
GROUP BYSELECT city, COUNT(*) FROM users GROUP BY city;Rows ko groups mein combine karta hai (aggregate functions ke saath use hota hai)
HAVINGGROUP BY city HAVING COUNT(*)>5;GROUP BY ke baad filter lagane ke liye (WHERE groups pe kaam nahi karta)

2. Filtering & Conditions

KeywordExampleKya karta hai
WHERESELECT * FROM users WHERE id=1;Condition lagata hai — sirf matching rows return honge
LIKEWHERE name LIKE 'A%';Pattern matching — text ke andar dhundta hai
%WHERE name LIKE '%raj%';LIKE ke saath wildcard — koi bhi/zero characters represent karta hai
_WHERE name LIKE 'r_j';LIKE ke saath wildcard — exactly ek character represent karta hai
INWHERE city IN ('Delhi','Mumbai');Multiple possible values check karta hai (OR ka shortcut)
NOT INWHERE city NOT IN ('Delhi','Mumbai');In list ke alawa sab kuch match karta hai
BETWEENWHERE age BETWEEN 18 AND 30;Range ke andar values check karta hai (inclusive)
IS NULLWHERE email IS NULL;Empty/missing value check karta hai
IS NOT NULLWHERE email IS NOT NULL;Value present hai, empty nahi
EXISTSWHERE EXISTS (SELECT 1 FROM orders WHERE user_id=users.id);Subquery ka result exist karta hai ya nahi check karta hai
ANY / SOMEWHERE age > ANY (SELECT age FROM users);Subquery ki kisi bhi ek value se condition true honi chahiye
ALLWHERE age > ALL (SELECT age FROM users);Subquery ki sabhi values se condition true honi chahiye

5. Joins (Multiple Tables Combine Karna)

KeywordExampleKya karta hai
JOIN / INNER JOINSELECT * FROM a JOIN b ON a.id=b.id;Dono tables mein matching rows hi return karta hai
LEFT JOINSELECT * FROM a LEFT JOIN b ON a.id=b.id;Left table ke sab rows + matching right rows (na milne pe NULL)
RIGHT JOINSELECT * FROM a RIGHT JOIN b ON a.id=b.id;Right table ke sab rows + matching left rows
FULL JOINSELECT * FROM a FULL JOIN b ON a.id=b.id;Dono tables ke sab rows, match ho ya na ho
SELF JOINSELECT a.name FROM users a JOIN users b ON a.mgr_id=b.id;Table ko khud ke saath join karna (hierarchical data ke liye)
CROSS JOINSELECT * FROM a CROSS JOIN b;Dono tables ka har combination (Cartesian product)
ONJOIN b ON a.id=b.user_id;Join ki condition specify karta hai — kaunse columns match karne hain
USINGJOIN b USING(id);ON ka shortcut jab dono tables mein column ka naam same ho
UNIONSELECT a FROM t1 UNION SELECT b FROM t2;Do queries ke results ko combine karta hai, duplicates remove karke
UNION ALLSELECT a FROM t1 UNION ALL SELECT b FROM t2;Same as UNION but duplicates bhi rakhta hai (faster)
INTERSECTSELECT a FROM t1 INTERSECT SELECT b FROM t2;Sirf wo rows deta hai jo dono queries mein common hain
EXCEPT / MINUSSELECT a FROM t1 EXCEPT SELECT b FROM t2;Pehli query ke rows jo dusri query mein nahi hain

6. Aggregate Functions

FunctionExampleKya karta hai
COUNT()SELECT COUNT(*) FROM users;Total rows ginta hai
SUM()SELECT SUM(price) FROM orders;Column ki values ka total nikalta hai
AVG()SELECT AVG(age) FROM users;Average nikalta hai
MAX()SELECT MAX(age) FROM users;Sabse badi value
MIN()SELECT MIN(age) FROM users;Sabse choti value
GROUP_CONCAT()SELECT GROUP_CONCAT(name) FROM users;MySQL — multiple rows ko ek single string mein jod deta hai
STRING_AGG()SELECT STRING_AGG(name,',') FROM users;PostgreSQL/MSSQL version of GROUP_CONCAT
ROW_NUMBER()SELECT ROW_NUMBER() OVER(ORDER BY id) FROM users;Har row ko sequential number deta hai (window function)
RANK() / DENSE_RANK()SELECT RANK() OVER(ORDER BY score DESC) FROM users;Rows ko rank deta hai, ties handle karne ke different tarike se
